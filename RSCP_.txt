                 	inviabilidades: 1 a cada período de 7 dias consecutivos
        	20: numero máximo de maquinistas; 
        	dist = somatório da "distancia" percorrida por todos os maquinistas;

+ w5 . ((total_dia_ferias_maquinistas - dias folga)/total_dia_ferias_maquinistas) +1)

		dias_folga: total de dias de folga/férias
